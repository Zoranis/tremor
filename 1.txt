# Replay speed: wall-clock seconds per 15-minute simulated slice.
# Default 0.45 → 672 slices (7 sim-days) per ~5 wall-min. Raise for slo-mo
# demos, lower if your pipeline can keep up.
REPLAY_SECONDS_PER_SLICE=0.45

# Inclusive window of GDELT 2.0 to replay. GDELT 2.0 publishes from
# 2015-02-18 onward; the default keeps a clean 7-day slice in early 2024.
# NOTE: data-init must download 672 slices × 3 files = ~2,000 HTTP requests
# on first run. With 16-way parallelism (DATA_INIT_WORKERS=16) expect 1–3
# wall-minutes on a good connection. Narrow to a single sim-day (96 slices)
# for an even faster smoke run:
#   REPLAY_WINDOW_START=2024-01-01
#   REPLAY_WINDOW_END=2024-01-01
REPLAY_WINDOW_START=2024-01-01
REPLAY_WINDOW_END=2024-01-08

# Concurrent download workers for data-init. 16 is a good default for
# GDELT's CDN; raise cautiously (rate-limit risk above ~32).
DATA_INIT_WORKERS=16

# When true, replay loops back to REPLAY_WINDOW_START after reaching END.
REPLAY_LOOP=true
